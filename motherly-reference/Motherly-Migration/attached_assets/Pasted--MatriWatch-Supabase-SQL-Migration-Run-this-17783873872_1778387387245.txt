-- ============================================
-- ENUMS
-- ============================================

create type user_role as enum ('mother', 'clinic_staff', 'admin');
create type risk_level as enum ('low', 'mid', 'high');


-- ============================================
-- TABLES
-- ============================================

-- Users
create table users (
  id uuid primary key default gen_random_uuid(),
  email text unique not null,
  role user_role not null default 'mother',
  created_at timestamptz default now()
);

-- Clinics
create table clinics (
  id uuid primary key default gen_random_uuid(),
  name text not null,
  location text,
  contact text,
  created_at timestamptz default now()
);

-- Mothers
create table mothers (
  id uuid primary key default gen_random_uuid(),
  user_id uuid references users(id) on delete cascade,
  name text not null,
  age int not null,
  gestational_age int,
  clinic_id uuid references clinics(id) on delete set null,
  due_date date,
  created_at timestamptz default now()
);

-- Staff
create table staff (
  id uuid primary key default gen_random_uuid(),
  user_id uuid references users(id) on delete cascade,
  clinic_id uuid references clinics(id) on delete set null,
  name text not null,
  role text,
  created_at timestamptz default now()
);

-- Check-ins
create table checkins (
  id uuid primary key default gen_random_uuid(),
  mother_id uuid references mothers(id) on delete cascade,
  bp_systolic int,
  bp_diastolic int,
  blood_sugar float,
  body_temp float,
  heart_rate int,
  symptoms text[],
  notes text,
  risk_score float,
  risk_level risk_level,
  created_at timestamptz default now()
);

-- EPDS Responses
create table epds_responses (
  id uuid primary key default gen_random_uuid(),
  mother_id uuid references mothers(id) on delete cascade,
  responses jsonb not null,
  total_score int not null,
  ppd_flagged boolean default false,
  created_at timestamptz default now()
);

-- Alerts
create table alerts (
  id uuid primary key default gen_random_uuid(),
  mother_id uuid references mothers(id) on delete cascade,
  clinic_id uuid references clinics(id) on delete cascade,
  alert_type text not null,
  message text not null,
  is_read boolean default false,
  created_at timestamptz default now()
);


-- ============================================
-- INDEXES
-- ============================================

create index idx_mothers_clinic_id on mothers(clinic_id);
create index idx_mothers_user_id on mothers(user_id);
create index idx_checkins_mother_id on checkins(mother_id);
create index idx_checkins_created_at on checkins(created_at desc);
create index idx_alerts_clinic_id on alerts(clinic_id);
create index idx_alerts_is_read on alerts(is_read);
create index idx_epds_mother_id on epds_responses(mother_id);
create index idx_staff_clinic_id on staff(clinic_id);
create index idx_staff_user_id on staff(user_id);

-- ============================================
-- RLS POLICIES — USERS
-- ============================================

-- Users can read their own row
create policy "users_select_own"
  on users for select
  using (auth.uid() = id);

-- Users can update their own row
create policy "users_update_own"
  on users for update
  using (auth.uid() = id);


-- ============================================
-- RLS POLICIES — CLINICS
-- ============================================

-- Anyone authenticated can read clinics
create policy "clinics_select_authenticated"
  on clinics for select
  using (auth.role() = 'authenticated');

-- Only admins can insert/update clinics
create policy "clinics_insert_admin"
  on clinics for insert
  with check (
    exists (
      select 1 from users
      where id = auth.uid() and role = 'admin'
    )
  );


-- ============================================
-- RLS POLICIES — MOTHERS
-- ============================================

-- Mothers can read their own record
create policy "mothers_select_own"
  on mothers for select
  using (user_id = auth.uid());

-- Clinic staff can read mothers assigned to their clinic
create policy "mothers_select_clinic_staff"
  on mothers for select
  using (
    exists (
      select 1 from staff
      where staff.user_id = auth.uid()
      and staff.clinic_id = mothers.clinic_id
    )
  );

-- Mothers can update their own record
create policy "mothers_update_own"
  on mothers for update
  using (user_id = auth.uid());

-- Mothers can insert their own record
create policy "mothers_insert_own"
  on mothers for insert
  with check (user_id = auth.uid());


-- ============================================
-- RLS POLICIES — STAFF
-- ============================================

-- Staff can read their own record
create policy "staff_select_own"
  on staff for select
  using (user_id = auth.uid());

-- Staff can read other staff in the same clinic
create policy "staff_select_same_clinic"
  on staff for select
  using (
    clinic_id in (
      select clinic_id from staff where user_id = auth.uid()
    )
  );


-- ============================================
-- RLS POLICIES — CHECKINS
-- ============================================

-- Mothers can read their own check-ins
create policy "checkins_select_own_mother"
  on checkins for select
  using (
    exists (
      select 1 from mothers
      where mothers.id = checkins.mother_id
      and mothers.user_id = auth.uid()
    )
  );

-- Mothers can insert their own check-ins
create policy "checkins_insert_own_mother"
  on checkins for insert
  with check (
    exists (
      select 1 from mothers
      where mothers.id = checkins.mother_id
      and mothers.user_id = auth.uid()
    )
  );

-- Clinic staff can read check-ins for their clinic's mothers
create policy "checkins_select_clinic_staff"
  on checkins for select
  using (
    exists (
      select 1 from mothers
      join staff on staff.clinic_id = mothers.clinic_id
      where mothers.id = checkins.mother_id
      and staff.user_id = auth.uid()
    )
  );

-- Service role can update check-ins (for risk score writing)
create policy "checkins_update_service_role"
  on checkins for update
  using (auth.role() = 'service_role');


-- ============================================
-- RLS POLICIES — EPDS RESPONSES
-- ============================================

-- Mothers can read their own EPDS responses
create policy "epds_select_own_mother"
  on epds_responses for select
  using (
    exists (
      select 1 from mothers
      where mothers.id = epds_responses.mother_id
      and mothers.user_id = auth.uid()
    )
  );

-- Mothers can insert their own EPDS responses
create policy "epds_insert_own_mother"
  on epds_responses for insert
  with check (
    exists (
      select 1 from mothers
      where mothers.id = epds_responses.mother_id
      and mothers.user_id = auth.uid()
    )
  );

-- Clinic staff can read EPDS responses for their clinic's mothers
create policy "epds_select_clinic_staff"
  on epds_responses for select
  using (
    exists (
      select 1 from mothers
      join staff on staff.clinic_id = mothers.clinic_id
      where mothers.id = epds_responses.mother_id
      and staff.user_id = auth.uid()
    )
  );


-- ============================================
-- RLS POLICIES — ALERTS
-- ============================================

-- Clinic staff can read alerts for their clinic
create policy "alerts_select_clinic_staff"
  on alerts for select
  using (
    exists (
      select 1 from staff
      where staff.user_id = auth.uid()
      and staff.clinic_id = alerts.clinic_id
    )
  );

-- Clinic staff can update alerts (mark as read)
create policy "alerts_update_clinic_staff"
  on alerts for update
  using (
    exists (
      select 1 from staff
      where staff.user_id = auth.uid()
      and staff.clinic_id = alerts.clinic_id
    )
  );

-- Service role can insert alerts (triggered by edge function)
create policy "alerts_insert_service_role"
  on alerts for insert
  with check (auth.role() = 'service_role');
